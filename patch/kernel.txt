#if DOF_TECHNIQUE > 0 && WORLD_BLUR == 2
    /*
        =====================================================================
        Optional DoF algorithm: Sundial Lite 1.2.0 (GPL-3.0)
        ---------------------------------------------------------------------
        Source of the algorithm (not copied verbatim, restructured for this
        pack's single DoF pass):
            shaders/programs/composite/Composite7.frag   CoC + CoC spread
            shaders/programs/composite/Composite8.frag   scatter-as-gather
        https://github.com/GeForceLegend/Sundial-Lite
        Sundial Lite is distributed under the GNU General Public License
        v3.0; this block is a derivative work and is covered by GPL-3.0.

        Sundial splits DoF over three passes (depth prep -> CoC spread ->
        gather). This pack only owns one full resolution pass before bloom,
        so the CoC spread of Sundial's Composite7 is evaluated per pixel here
        and immediately consumed by the Composite8 style gather. That keeps
        the result close to Sundial while staying inside composite3.

        Sign convention of the CoC (same as Sundial): negative in front of
        the focal plane, positive behind it. The gather compares distances
        instead of the signed CoC, which is equivalent and easier to read.
        =====================================================================
    */
    const float dofGoldenAngle = 2.39996323;    // Sundial Common.glsl: goldenRotate
    const float dofRadiusUnit = 15.0 / 1080.0;  // Sundial: 15 screen pixels per CoC unit, kept relative to the view height

    float DoFSundialWindowZ(float dist) {
        return far * (dist - near) / (dist * (far - near));
    }

    float DoFSundialDistFromZ(float z) {
        z = clamp(z, 0.0, 0.9999999);
        return max(near * far / (far - z * (far - near)), 0.3);
    }

    float DoFSundialLinearDepth(vec2 coord) {
        // Same depth source as the Legacy DoF (depthtex1), plus the LOD
        // fallback the pack needs for Distant Horizons / Voxy. Voxy keeps
        // excludeLodsFromVanillaDepth=true, so LOD terrain is not in
        // depthtex1 at all and has to come from its own depth buffer.
        vec2 minCoord = 0.5 / vec2(viewWidth, viewHeight);
        coord = clamp(coord, minCoord, vec2(1.0) - minCoord);
        ivec2 dofTexel = ivec2(coord * vec2(viewWidth, viewHeight));
        float z = texelFetch(depthtex1, dofTexel, 0).r;
        if (z < 0.56) return far; // Hand (and anything closer than ~11 cm, same cut as the Legacy DoF)
        #if defined DISTANT_HORIZONS || defined VOXY
            if (z >= 1.0) {
                #ifdef DISTANT_HORIZONS
                    float lodZ = texelFetch(dhDepthTex1, dofTexel, 0).r;
                    vec4 screenPosLod = vec4(coord, lodZ, 1.0);
                    vec4 viewPosLod = dhProjectionInverse * (screenPosLod * 2.0 - 1.0);
                #else
                    float lodZ = texelFetch(vxDepthTexOpaque, dofTexel, 0).r;
                    vec4 screenPosLod = vec4(coord, lodZ, 1.0);
                    vec4 viewPosLod = vxProjInv * (screenPosLod * 2.0 - 1.0);
                #endif
                viewPosLod /= viewPosLod.w;
                return max(-viewPosLod.z, 0.3);
            }
        #endif
        return DoFSundialDistFromZ(z);
    }

    float DoFSundialCoC(float focus, float dist, float fovScale) {
        // Sundial Composite7 circleOfConfusionRadius(): physical thin lens.
        //   c = aperture * (|d| - focus) / (|d| * (focus - focalLength))
        // "Depth of Field Strength" scales the aperture so the existing
        // strength slider keeps working (64.0 = Sundial's own aperture).
        float denominator = max(dist, 1e-3) * max(focus - DOF_FOCAL_LENGTH, 1e-4);
        float coc = (dist - focus) / denominator;
        coc *= DOF_APERTURE_DIAMETER_SCALE * (WB_DOF_I / 64.0) * fovScale;
        return clamp(coc, -DOF_MAX_BLUR_RADIUS, DOF_MAX_BLUR_RADIUS);
    }

    float DoFSundialFocusTilted(float focus, vec2 coord) {
        // Legacy tilt-shift adds an offset to the window space depth of the
        // focal plane. Doing the same here and converting back to a distance
        // keeps the tilt direction and strength identical to Legacy, but the
        // focus plane now feeds the physical CoC instead of a fixed blur.
        #if TILT_SHIFT != 0 || TILT_SHIFT2 != 0
            float tiltOffset = 0.0;
            #if TILT_SHIFT != 0
                tiltOffset += (coord.y * 2.0 - 1.0) * 0.0007 * float(TILT_SHIFT);
            #endif
            #if TILT_SHIFT2 != 0
                tiltOffset += (coord.x * 2.0 - 1.0) * 0.0007 * float(TILT_SHIFT2);
            #endif
            return DoFSundialDistFromZ(DoFSundialWindowZ(focus) - tiltOffset);
        #else
            return focus;
        #endif
    }

    vec2 DoFSundialNoise() {
        // Two decorrelated per pixel values, shifted every frame so TAA can
        // resolve the spiral sampling. Sundial uses its blue noise texture
        // here; this pack's noisetex has a different layout, so the same job
        // is done with a hash.
        vec2 p = gl_FragCoord.xy;
        vec2 noise = vec2(
            fract(sin(dot(p, vec2(12.9898, 78.233))) * 43758.5453),
            fract(sin(dot(p, vec2(39.3468, 11.1351))) * 24634.6345)
        );
        return fract(noise + vec2(0.61803398875, 0.41421356237) * mod(float(frameCounter), 64.0));
    }

    vec2 DoFSundialDiskOffset(vec2 dir, float radius) {
        // radius is in view height units; x is corrected so the disk stays
        // round on wide screens (Sundial works in pixels, same result).
        vec2 offset = dir * radius;
        offset.x /= aspectRatio;
        return offset;
    }

    void DoWorldBlurSundial(inout vec3 color, float z1) {
        if (z1 < 0.56) return; // Hand

        const int spreadSamples = DOF_COC_SPREAD_SAMPLES;
        const int gatherSamples = DOF_SAMPLES;

        float fovScale = 1.0;
        #ifdef WB_FOV_SCALED
            fovScale = gbufferProjection[1][1] * 0.8;
        #endif

        float centerDist = DoFSundialLinearDepth(texCoord);
        #if WB_DOF_FOCUS == 0
            float minFocus = max(DOF_FOCAL_LENGTH + 0.01, 0.3);
            float focusDist = max(DoFSundialDistFromZ(centerDepthSmooth), minFocus);
        #elif WB_DOF_FOCUS == -1
            float focusDist = max(200.0 * pow2(vsBrightness), 0.3);
        #else
            float focusDist = float(WB_DOF_FOCUS);
        #endif

        float focusEff = DoFSundialFocusTilted(focusDist, texCoord);
        float centerCoC = DoFSundialCoC(focusEff, centerDist, fovScale);

        vec2 noise = DoFSundialNoise();
        mat2 rotation = mat2(cos(dofGoldenAngle), sin(dofGoldenAngle), -sin(dofGoldenAngle), cos(dofGoldenAngle));
        vec2 dir = vec2(cos(noise.x * tau), sin(noise.x * tau));
        float maxRadius = abs(centerCoC);

        // CoC spread (Sundial Composite7): foreground bokeh has to reach over
        // the sharp background, so the largest CoC of everything in front of
        // this pixel becomes the gather radius.
        float stepRadius = noise.y / float(spreadSamples);
        for (int i = 0; i < spreadSamples; i++) {
            float radius = sqrt(stepRadius) * DOF_MAX_BLUR_RADIUS;
            vec2 sampleCoord = texCoord + DoFSundialDiskOffset(dir, radius * dofRadiusUnit);
            float sampleDist = DoFSundialLinearDepth(sampleCoord);
            float sampleCoC = abs(DoFSundialCoC(DoFSundialFocusTilted(focusDist, sampleCoord), sampleDist, fovScale));
            if (sampleCoC > radius && sampleDist <= centerDist) maxRadius = max(maxRadius, sampleCoC);
            dir = rotation * dir;
            stepRadius += 1.0 / float(spreadSamples);
        }

        if (maxRadius * dofRadiusUnit * viewHeight > 0.5) {
            // Gather (Sundial Composite8): samples that this pixel's own disk
            // can cover keep their own colour, samples in front that can cover
            // this pixel are blended in, everything else only counts as weight.
            vec3 totalColor = vec3(0.0);
            float totalSamples = 0.0;
            vec3 selfColor = color * 1e-5;
            float selfSamples = 1e-5;
            float selfWeight = 0.0;

            float spreadNorm = maxRadius / DOF_MAX_BLUR_RADIUS; // Sundial normalises the radius to 0..1
            float centerWeight = spreadNorm * spreadNorm - spreadNorm + 1.0;
            centerWeight = mix(centerWeight, 1e-5, pow(abs(centerCoC) / max(maxRadius, 1e-5), 4.0));

            #ifdef WB_CHROMATIC
                vec2 midDist = texCoord - 0.5;
                vec2 chromaticScale = sign(midDist) * sqrt(abs(midDist)) * vec2(1.0, viewHeight / viewWidth);
                vec2 aberration = (15.0 / vec2(viewWidth, viewHeight)) * chromaticScale * (abs(centerCoC) / DOF_MAX_BLUR_RADIUS);
            #endif

            vec2 minCoord = 0.5 / vec2(viewWidth, viewHeight);
            vec2 maxCoord = vec2(1.0) - minCoord;
            dir = vec2(cos(noise.x * tau), sin(noise.x * tau));
            float stepGather = noise.y / float(gatherSamples);
            for (int i = 0; i < gatherSamples; i++) {
                float radius = sqrt(stepGather) * maxRadius;
                vec2 offset;
                #if DOF_SHAPE == 0
                    offset = DoFSundialDiskOffset(dir, radius * dofRadiusUnit);
                    #ifdef WB_ANAMORPHIC
                        offset.y *= 1.5;
                    #endif
                #else
                    offset = getPolygonOffset(atan(dir.y, dir.x), DOF_SHAPE, radius * dofRadiusUnit);
                    offset.x /= aspectRatio;
                #endif

                vec2 sampleCoord = clamp(texCoord + offset, minCoord, maxCoord);
                #ifdef WB_CHROMATIC
                    vec2 sampleCoordR = clamp(texCoord + offset + aberration, minCoord, maxCoord);
                    vec2 sampleCoordB = clamp(texCoord + offset - aberration, minCoord, maxCoord);
                    vec3 sampleColor = vec3(texture2DLod(colortex0, sampleCoordR, 0.0).r,
                                            texture2DLod(colortex0, sampleCoord,  0.0).g,
                                            texture2DLod(colortex0, sampleCoordB, 0.0).b);
                #else
                    vec3 sampleColor = texture2DLod(colortex0, sampleCoord, 0.0).rgb;
                #endif

                float sampleDist = DoFSundialLinearDepth(sampleCoord);
                float sampleCoC = abs(DoFSundialCoC(DoFSundialFocusTilted(focusDist, sampleCoord), sampleDist, fovScale));

                if (abs(centerCoC) >= radius && sampleDist >= centerDist) {
                    selfColor += sampleColor;
                    selfSamples += 1.0;
                } else if (sampleCoC >= radius && sampleDist <= centerDist) {
                    float weight = max(1.0, abs(centerCoC) / max(sampleCoC, 1e-5));
                    weight *= weight;
                    totalColor += sampleColor * weight;
                    totalSamples += weight;
                } else {
                    selfWeight += centerWeight;
                }

                dir = rotation * dir;
                stepGather += 1.0 / float(gatherSamples);
            }

            totalColor += selfColor * (1.0 + selfWeight / selfSamples);
            totalColor /= totalSamples + selfSamples + selfWeight;

            #ifdef DOF_SNEAKING
                color = mix(color, totalColor, DoFSneaking);
            #else
                color = totalColor;
            #endif
        }

        #ifdef DOF_VIEW_FOCUS_NEW2
            vec3 cocColor = vec3(1.0, 0.0, 0.8) * 10.0;
            float cocView = float(abs(centerCoC) < 0.0025 && isDoFGUI);
            #ifdef DOF_SNEAKING
                color = mix(color, cocColor, cocView * DoFSneaking);
            #else
                color = mix(color, cocColor, cocView);
            #endif
        #endif
    }
#endif
